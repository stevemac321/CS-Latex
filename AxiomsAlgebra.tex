\section*{Axioms of Algebra}

\begin{itemize}
    \item \textbf{Closure Law}:
        \begin{itemize}
            \item For Addition: \( a + b \in \mathbb{R} \) for all \( a, b \in \mathbb{R} \).
            \item For Multiplication: \( a \cdot b \in \mathbb{R} \) for all \( a, b \in \mathbb{R} \).
        \end{itemize}
    \item \textbf{Commutative Law}:
        \begin{itemize}
            \item For Addition: \( a + b = b + a \).
            \item For Multiplication: \( a \cdot b = b \cdot a \).
        \end{itemize}
    \item \textbf{Associative Law}:
        \begin{itemize}
            \item For Addition: \( (a + b) + c = a + (b + c) \).
            \item For Multiplication: \( (a \cdot b) \cdot c = a \cdot (b \cdot c) \).
        \end{itemize}
    \item \textbf{Distributive Law}:
        \[
        a \cdot (b + c) = a \cdot b + a \cdot c
        \]
    \item \textbf{Identity Law}:
        \begin{itemize}
            \item For Addition: \( a + 0 = a \).
            \item For Multiplication: \( a \cdot 1 = a \).
        \end{itemize}
    \item \textbf{Inverse Law}:
        \begin{itemize}
            \item For Addition: \( a + (-a) = 0 \).
            \item For Multiplication: \( a \cdot \frac{1}{a} = 1 \) (if \( a \neq 0 \)).
        \end{itemize}
\end{itemize}
